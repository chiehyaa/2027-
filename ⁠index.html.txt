    // Initialize UI
    function init() {
      // 1. Render Day Chips
      const chipContainer = document.getElementById('day-chips');
      daysData.forEach((d, idx) => {
        const chip = document.createElement('div');
        chip.className = `day-chip ${idx === 0 ? 'active' : ''}`;
        chip.innerText = `D${d.day} ${d.date.split(' ')[0]}`;
        chip.onclick = () => renderDay(idx);
        chipContainer.appendChild(chip);
      });

      // 2. Render Day Details
      renderDay(0);

      // 3. Render Hotels
      const hotelList = document.getElementById('hotel-list');
      hotelsData.forEach(h => {
        const item = document.createElement('div');
        item.style.marginBottom = '12px';
        item.style.paddingBottom = '10px';
        item.style.borderBottom = '1px dashed var(--border)';
        item.innerHTML = `
          <div style="font-weight:700; color:var(--primary); font-size:0.9rem;">${h.city} ｜ ${h.name}</div>
          <div style="font-size:0.75rem; color:var(--text-muted); margin:2px 0;">📍 ${h.address}</div>
          <div style="font-size:0.8rem; color:#059669;">🛡️ 安全亮點：${h.note}</div>
          <a class="btn-map" href="https://www.google.com/maps/search/?api=1&query=${encodeURIComponent(h.name + ' ' + h.address)}" target="_blank">Google Map 導航</a>
        `;
        hotelList.appendChild(item);
      });

      // 4. Render Packing Checklist
      const packList = document.getElementById('packing-checklist');
      packingItems.forEach((text, i) => {
        const item = document.createElement('label');
        item.className = 'checklist-item';
        item.innerHTML = `
          <input type="checkbox" onchange="toggleCheck(this)" />
          <span>${text}</span>
        `;
        packList.appendChild(item);
      });
    }

    function renderDay(index) {
      document.querySelectorAll('.day-chip').forEach((c, idx) => {
        c.classList.toggle('active', idx === index);
      });

      const d = daysData[index];
      const container = document.getElementById('day-details');
      let scheduleHtml = '';
      d.schedule.forEach(s => {
        scheduleHtml += `
          <div class="timeline-item">
            <div class="timeline-time">${s.time}</div>
            <div class="timeline-desc">${s.desc}</div>
            ${s.map ? `<a class="btn-map" href="https://www.google.com/maps/search/?api=1&query=${encodeURIComponent(s.map)}" target="_blank">📍 開啟導航：${s.map.split(' ')[0]}</a>` : ''}
          </div>
        `;
      });

      container.innerHTML = `
        <div class="card">
          <div class="card-title">
            <span>Day ${d.day} ｜ ${d.date}</span>
            <span class="badge blue">${d.badge}</span>
          </div>
          <div style="font-weight:700; margin-bottom:14px; font-size:0.95rem;">${d.title}</div>
          ${scheduleHtml}
          <div style="margin-top:14px; padding-top:10px; border-top:1px solid var(--border); font-size:0.8rem; display:flex; justify-content:space-between; color:var(--text-muted);">
            <span>預估花費：<strong style="color:var(--accent);">${d.cost}</strong></span>
            <span>下榻：<strong>${d.hotel}</strong></span>
          </div>
        </div>
      `;
    }

    function toggleCheck(cb) {
      const span = cb.nextElementSibling;
      if (cb.checked) {
        span.style.textDecoration = 'line-through';
        span.style.opacity = '0.5';
      } else {
        span.style.textDecoration = 'none';
        span.style.opacity = '1';
      }
    }

    window.onload = init;
  </script>
